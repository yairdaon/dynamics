\chapter{Linear Algebra}

\section{Matrix-vector multiplication}

A $2\times 2$ matrix $A=\begin{pmatrix}a&b\\c&d\end{pmatrix}$ acts on
a vector $v=\begin{pmatrix}x\\y\end{pmatrix}$ by
\[
Av = x\begin{pmatrix}a\\c\end{pmatrix} + y\begin{pmatrix}b\\d\end{pmatrix}
= \begin{pmatrix}ax+by\\cx+dy\end{pmatrix}.
\]
That is, $Av$ is the combination of $A$'s two columns,
$\begin{pmatrix}a\\c\end{pmatrix}$ and
  $\begin{pmatrix}b\\d\end{pmatrix}$, weighted by $v$'s entries $x,y$.
   
\section{The determinant as an area}

Take two vectors in the plane, $u=(a,c)$ and $v=(b,d)$. They span a
parallelogram, see figure below.

\begin{center}
\begin{tikzpicture}[scale=0.9]
  \draw[->] (0,0) -- (4.5,0);
  \draw[->] (0,0) -- (0,3.5);
  \coordinate (O) at (0,0);
  \coordinate (U) at (3,1);
  \coordinate (V) at (1,1.8);
  \coordinate (W) at ($(U)+(V)$);
  \draw[thick,-{Stealth[]}] (O) -- (U) node[below right] {$u=(a,c)$};
  \draw[thick,-{Stealth[]}] (O) -- (V) node[above left] {$v=(b,d)$};
  \draw[dashed] (U) -- (W) -- (V);
  \draw[fill=gray!20,opacity=0.5] (O)--(U)--(W)--(V)--cycle;
  \draw[dotted] (0,0) rectangle (4,2.8);
\end{tikzpicture}
\end{center}

Enclose the parallelogram in the axis-aligned bounding box of width
$a+b$ and height $c+d$. The box consists of the parallelogram, two
right triangles with legs $a,c$ (combined area $ac$), two right
triangles with legs $b,d$ (combined area $bd$), and two $b\times c$
rectangles (combined area $2bc$). So
\[
(a+b)(c+d) = \text{Area} + ac + bd + 2bc
\quad\Longrightarrow\quad
\text{Area} = ad - bc.
\]

\begin{definition}
The \emph{determinant} of a matrix
$A=\begin{pmatrix}a&b\\c&d\end{pmatrix}$ is the \emph{signed} area
spanned by $u=(a,c)$, $v=(b,d)$. 
\end{definition}
\subsubsection{Facts regarding determinants}
\begin{enumerate}
\item We had just shown $\det A = ad-bc$.

\item From 1, when $0 = \det A = ad-bc$, define the ratio $\alpha =
  \frac{a}{b} = \frac{c}{d}$, so that
\[
\begin{pmatrix}a\\c\end{pmatrix} = \alpha \begin{pmatrix}b\\d\end{pmatrix}.
\]
I.e. when $\det A = 0$ the columns of $A$ are parallel.

\item From 1:
\[
\det \begin{pmatrix}a&b\\c&d\end{pmatrix} = ad-bc = ad-cb =
\det \begin{pmatrix}a&c\\b&d\end{pmatrix}.
\]
 \item Therefore, from 2 and 3: $A$'s columns are parallel if and only
   if $A$'s rows are parallel.
\end{enumerate}
\section{Matrix inverse}
The \emph{inverse} of $A$ is denoted $A^{-1}$ and has the properties:
\begin{itemize}
\item $AA^{-1} = I$.
\item $A^{-1} A = I$.
\item $Ax = b \Longleftrightarrow x = A^{-1}b$.
\end{itemize}
Only matrices with $\det A \neq 0$ have an inverse.

\section{Eigenvalues of a $2\times 2$ matrix}

\begin{definition}
A nonzero vector $v$ is an \emph{eigenvector} of $A$ with
\emph{eigenvalue} $\lambda$ if $Av = \lambda v$.
\end{definition}

Rearranging, $(A-\lambda I)v = 0$ for some $v\ne 0$. We saw that
$(A-\lambda I)v$ is the combination of $(A-\lambda I)$'s two columns
weighted by $v$'s entries. This is only possible $(A-\lambda I)$'s
columns are parallel, i.e.
\[
\det(A-\lambda I) = 0.
\]

\paragraph{Useful fact.}
If $\lambda_1, \lambda_2$ are $A$'s eigenvalues, then $\det A=
\lambda_1 \lambda_2$ and $\text{tr} A = a+d = \lambda_1 + \lambda_2$.

\section{Eigendecomposition}
Suppose $A$ has eigenvectors $v_1,v_2$ with eigenvalues
$\lambda_1,\lambda_2$, and let $P=\begin{pmatrix}v_1 & v_2\end{pmatrix}$
(eigenvectors as columns). Then
\[
AP = \begin{pmatrix}Av_1 & Av_2\end{pmatrix}
   = \begin{pmatrix}\lambda_1 v_1 & \lambda_2 v_2\end{pmatrix}
   = PD, \qquad D=\begin{pmatrix}\lambda_1&0\\0&\lambda_2\end{pmatrix},
\]
so, if $v_1,v_2$ are not parallel (so $P$ is invertible),
\[
A = PDP^{-1}.
\]
$P$ is a \emph{change of basis}: $P^{-1}$ converts a vector from the
standard basis to \emph{eigenbasis} coordinates (coefficients along
$v_1,v_2$), and $P$ converts back. In eigenbasis coordinates $A$ acts
as the diagonal $D$.

\section{Example}
Let
\[
A = \begin{pmatrix}1&2\\3&4\end{pmatrix}.
\]
Then $\det A = 1\cdot 4 - 2\cdot 3 = -2$ and $\operatorname{tr}A =
1+4=5$. By the useful fact above, $A$'s eigenvalues $\lambda_1,\lambda_2$
satisfy
\[
\lambda_1\lambda_2 = \det A = -2, \qquad \lambda_1+\lambda_2 =
\operatorname{tr}A = 5.
\]

\subsection*{Finding the eigenvalues}
From the first equation, $\lambda_1 = -2/\lambda_2$. Substituting into
the second and dropping subscripts (we are just solving for a number
$\lambda$ satisfying both constraints):
\[
-\frac{2}{\lambda} + \lambda = 5 \quad\Longrightarrow\quad \lambda^2 -
5\lambda - 2 = 0.
\]
The quadratic formula gives
\[
\lambda_{1,2} = \frac{5\pm\sqrt{33}}{2}.
\]
We can check directly that their product is indeed $-2$ and their sum is $5$. So
\[
\lambda_1 = \frac{5+\sqrt{33}}{2}, \qquad \lambda_2 =
\frac{5-\sqrt{33}}{2}.
\]

\subsection*{Finding the eigenvectors}
Fix an eigenvalue $\lambda$ (either $\lambda_1$ or $\lambda_2$) and
write
\[
A - \lambda I = \begin{pmatrix}1-\lambda & 2\\ 3 & 4-\lambda\end{pmatrix}.
\]
Writing $v=(x,y)$ explicitly, $(A-\lambda I)v=0$ is the pair of
equations
\begin{align*}
(1-\lambda)x + 2y &= 0, \\
3x+ (4-\lambda)y &= 0.
\end{align*}
Since $\det (A-\lambda I) = 0$, fact 4 above implies the rows of
$A-\lambda I$ are multiples of each other, hence one of the equations
is redundant. From the first equation,
\[
y = \frac{\lambda-1}{2}x.
\]
Taking $x=2$ gives the eigenvector
\[
v = \begin{pmatrix}2\\ \lambda-1\end{pmatrix},
\]
one for each of $\lambda=\lambda_1,\lambda_2$.

\section{The determinant for a $3\times 3$ matrix}
\[
\det\begin{pmatrix}a_1&b_1&c_1\\a_2&b_2&c_2\\a_3&b_3&c_3\end{pmatrix}
= a_1\det\begin{pmatrix}b_2&c_2\\b_3&c_3\end{pmatrix}
- a_2\det\begin{pmatrix}b_1&c_1\\b_3&c_3\end{pmatrix}
+ a_3\det\begin{pmatrix}b_1&c_1\\b_2&c_2\end{pmatrix}.
\]
The same interpretations hold: the determinant is (signed) volume, the
determinant is the product of eigenvalues, and the trace is the sum of
eigenvalues.
