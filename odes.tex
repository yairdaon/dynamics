\chapter{Ordinary Differential Equations}

\section{A single linear equation}
\[
\dot x = \lambda x.
\]
We had already seen such equation. It was the equation for $Y(t)$
without infections. The solution is identical to what we had.
\[
x(t) = x(0) e^{\lambda t}.
\]
If $\lambda>0$ then $x(t)$ grows exponentialy; if $\lambda<0$ then
$x(t)$ decays exponentialy.

\section{Uncoupled linear equations}
Now, consider two equations:
\begin{align}
  \dot x_1 &= \lambda_1 x_1 \\ % \qquad x_1(0)=x^{(1)}_0. \\
  \dot x_2 &= \lambda_2 x_2.   % \qquad x_2(0)=x^{(2)}_0.
\end{align}
The solutions are
\[
\begin{array}{r@{\ }c@{\ }l}
  \dot x_1 &=& \lambda_1 x_1 \\
  \dot x_2 &=& \lambda_2 x_2
\end{array}
\quad\Longrightarrow\quad
\begin{array}{r@{\ }c@{\ }l}
  x_1(t) &=& x_1(0)e^{\lambda_1 t} \\
  x_2(t) &=& x_2(0)e^{\lambda_2 t}
\end{array}
\]
We can depict solutions of this set of equations. The qualitative
picture depends on the signs of $\lambda_1$ and $\lambda_2$, see
Figure~\ref{fig:odes}.

\begin{figure}[h]
\centering
\includegraphics[width=\linewidth]{visualizations/phase_portraits_real.pdf}
\caption{Both negative: every trajectory decays to the origin
  (\emph{stable node}). Both positive: every trajectory grows away
  from it (\emph{unstable node}). Opposite signs: the origin attracts
  along one axis and repels along the other (\emph{saddle}).}\label{fig:odes}
\end{figure}


\subsection{Matrix form}
We can write the uncoupled set of equations using in matrix form via a
diagonal matrix:
\[
\begin{pmatrix}
  \dot x_1 \\
  \dot x_2
\end{pmatrix}
= \begin{pmatrix}
  \lambda_1 & 0 \\
  0 & \lambda_2
\end{pmatrix}
\begin{pmatrix}
  x_1 \\
  x_2
\end{pmatrix}.
\]



\section{Cycles}
Consider a simple coupled system
\begin{align}
  \dot x_1 &= x_2 \\
  \dot x_2 &= -x_1.
\end{align}
Recal the derivatives of sine and cosine, see
Figure~\ref{fig:sin-cos-derivatives} for an illustratio:
\[
\frac{d}{dx}\sin x = \cos x, \qquad \frac{d}{dx}\cos x = -\sin x.
\]
\begin{figure}[h]
\centering
\includegraphics[width=0.75\linewidth]{visualizations/sin_cos_derivatives.pdf}
\caption{$\sin x$ and $\cos x$ at $x=0,\pi/2,\pi,3\pi/2,2\pi$ (blue
  values), with the tangent line and its slope (red) at each
  point.}\label{fig:sin-cos-derivatives}
\end{figure}
Differentiating once more,
\begin{align}
  \ddot x_1 &= -x_1 \\
  \ddot x_2 &= -x_2.
\end{align}
Both $x_1,x_2$ solve $\ddot x = -x$, whose solutions are sines and
cosines. Specifically
\begin{align*}
x_1(t) &= x_1(0)\cos t + x_2(0)\sin t \\
x_2(t) &= x_2(0)\cos t - x_1(0)\sin t.
\end{align*}
Sines and cosines are \emph{periodic} functions, so the solution
should also be periodic. In fact, solutions to this system are cycles,
see Figure \ref{fig:cycles} middle.

\subsection{Diagonalizing}
We can also write this system in matrix form:
\[
\begin{pmatrix}\dot x_1 \\ \dot x_2\end{pmatrix}
= \underbrace{\begin{pmatrix}0&1\\-1&0\end{pmatrix}}_{J}
\begin{pmatrix}x_1\\x_2\end{pmatrix}.
\]
$J$'s eigenvalues solve $\det(J-\lambda I)=\lambda^2+1=0$, so
$\lambda=\pm i$ with $i = \sqrt{-1}$. This is not a coincidence: the
\emph{imaginary} part of an eigenvalue gives rise to \emph{cyclic}
behavior whereas the \emph{real} part governs growth and decay.

\section{(Almost) every linear system is diagonal, in the right coordinates}
Let $A$ have eigenvectors $v_1,v_2$ with corresponding eigenvalues
$\lambda_1,\lambda_2$. Then $AP = PD$ with
$P=\begin{pmatrix}v_1&v_2\end{pmatrix}$ and consequently
$D = P^{-1}AP$. Write $x = Pu$, i.e.\ $u = P^{-1}x$. Then
\begin{align*}
  \dot u &= P^{-1}\dot x\\
  &= P^{-1}Ax \\
  &= P^{-1}APu \\
  &= Du.
\end{align*}
So $u$ solves a \emph{diagonal} system. Its behavior depends on the
real and complex parts of the eigenvalues.
\begin{enumerate}
\item Positive real part $\Rightarrow$ exponential growth.
\item Negative real part $\Rightarrow$ exponential decay.
\item Nonzero imaginary part $\Rightarrow$ rotation.
\end{enumerate}

\subsection{Exercise 2}
Let
\[
A = \begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
Consider $\dot x = Ax$. Which of the three panels in
Figure~\ref{fig:odes} does this system belong to?


\paragraph{Solution.}
First, we find $A$'s eigenvalues and eigenvectors.  $\det A = 0\cdot 0
- 1\cdot 1 = -1$ and $\operatorname{tr}A = 0+0=0$, so
\[
\lambda_1\lambda_2 = -1, \qquad \lambda_1+\lambda_2 = 0.
\]
We can easily guess $\lambda_1=1,\ \lambda_2=-1$.

For $\lambda_1=1$: $(A-I)v=0$ with $v=(x,y)$ is
\begin{align*}
  -x+y&=0 \\
  x-y&=0.
\end{align*}
so $v_1=(1,1)$.
For $\lambda_2=-1$: $(A+I)v=0$ implies
\begin{align*}
  x+y&=0 \\
  x+y&=0
\end{align*}
so $v_2=(1,-1)$.

The eigenvalues $1,-1$ have opposite signs, so this system is a
\emph{saddle}. Trajectories on $v_1=(1,1)$ move away from the origin
while trajectories on $v_2=(1,-1)$ approach it. See
Figure~\ref{fig:exercise2} for a full phase portrait.

\subsection{Exercise 3}
Let
\[
B = \begin{pmatrix}1&-2\\2&1\end{pmatrix}.
\]
Consider $\dot x = Bx$. Which of the three types in
Figure~\ref{fig:cycles} does this system belong to?

\paragraph{Solution.}
$\det B = 1\cdot 1 - (-2)\cdot 2 = 5$ and $\operatorname{tr}B = 1+1=2$,
so $B$'s eigenvalues satisfy
\[
\lambda_1\lambda_2 = 5, \qquad \lambda_1+\lambda_2 = 2.
\]
From the second equation $\lambda_2=2-\lambda_1$; substituting into
the first and dropping subscripts, $\lambda(2-\lambda)=5$, i.e.\
$\lambda^2-2\lambda+5=0$. The quadratic formula gives
\[
\lambda = \frac{2\pm\sqrt{4-20}}{2} = 1\pm 2i.
\]
The eigenvalues are complex, with positive real part $1$ and imaginary
part $\pm 2$. By the real part, trajectories grow away from the
origin. The imaginary part implies trajectories circle the
origin. This system is a \emph{spiral source}.
