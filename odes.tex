\chapter{Ordinary Differential Equations}

\section{A single linear equation}
\[
\dot x = \lambda x.
\]
We had already seen such equation. It was the equation for $Y(t)$
without infections. The solution is identical.
\[
x(t) = x(0) e^{\lambda t}.
\]
If $\lambda>0$ we have exponential growth; if $\lambda<0$ we have
exponential decay.

\section{Uncoupled linear equations}
Now, consider two equations:
\begin{align}
  \dot x_1 &= \lambda_1 x_1 \\ % \qquad x_1(0)=x^{(1)}_0. \\
  \dot x_2 &= \lambda_2 x_2.   % \qquad x_2(0)=x^{(2)}_0.
\end{align}
The solutions are
\[
\begin{array}{r@{\ }c@{\ }l}
  \dot x_1 &=& \lambda_1 x_1 \\
  \dot x_2 &=& \lambda_2 x_2
\end{array}
\quad\Longrightarrow\quad
\begin{array}{r@{\ }c@{\ }l}
  x_1(t) &=& x_1(0)e^{\lambda_1 t} \\
  x_2(t) &=& x_2(0)e^{\lambda_2 t}
\end{array}
\]
We can depict solutions of this set of equations. The qualitative
picture depends on the signs of $\lambda_1$ and $\lambda_2$, see
Figure~\ref{fig:odes}.

\begin{figure}[h]
\centering
\includegraphics[width=\linewidth]{visualizations/phase_portraits_real.pdf}
\caption{Both negative: every trajectory decays to the origin
  (\emph{stable node}). Both positive: every trajectory grows away
  from it (\emph{unstable node}). Opposite signs: the origin attracts
  along one axis and repels along the other (\emph{saddle}).}\label{fig:odes}
\end{figure}
\subsection{Matrix form}
We can write this set of equations using a diagonal matrix:
\[
\begin{pmatrix}
  \dot x_1 \\
  \dot x_2
\end{pmatrix}
= \begin{pmatrix}
  \lambda_1 & 0 \\
  0 & \lambda_2
\end{pmatrix}
\begin{pmatrix}
  x_1 \\
  x_2
\end{pmatrix}.
\]

\section{Cycles}
Consider a simple coupled system
\begin{align}
  \dot x_1 &= x_2 \\
  \dot x_2 &= -x_1.
\end{align}
Differentiating once more,
\begin{align}
  \ddot x_1 &= -x_1 \\
  \ddot x_2 &= -x_2.
\end{align}
Both $x_1,x_2$ solve $\ddot x = -x$, whose solutions are sines and
cosines. Specifically
\begin{align*}
x_1(t) &= x_1(0)\cos t + x_2(0)\sin t \\
x_2(t) &= x_2(0)\cos t - x_1(0)\sin t.
\end{align*}
Sines and cosines are \emph{periodic} functions, so the solution
should also be periodic. In fact, solutions to this system are cycles,
see Figure \ref{fig:cycles} middle.

\subsection{Diagonalizing}
We can also write this system in matrix form:
\[
\begin{pmatrix}\dot x_1 \\ \dot x_2\end{pmatrix}
= \underbrace{\begin{pmatrix}0&1\\-1&0\end{pmatrix}}_{J}
\begin{pmatrix}x_1\\x_2\end{pmatrix}.
\]
$J$'s eigenvalues solve $\det(J-\lambda I)=\lambda^2+1=0$, so
$\lambda=\pm i$ with $i = \sqrt{-1}$. This is not a coincidence: the
\emph{imaginary} part of an eigenvalue gives rise to \emph{cyclic}
behavior whereas the \emph{real} part governs growth and decay.

\section{(Almost) every linear system is diagonal, in the right coordinates}
Let $A$ have eigenvectors $v_1,v_2$ with corresponding eigenvalues
$\lambda_1,\lambda_2$ (possibly complex). Then $AP = PD$ with
$P=\begin{pmatrix}v_1&v_2\end{pmatrix}$ and consequently
$A=PDP^{-1}$. Write $x = Pu$, i.e.\ $u = P^{-1}x$. Then
\begin{align*}
  \dot u &= P^{-1}\dot x\\
  &= P^{-1}Ax \\
  &= P^{-1}APu \\
  &= Du.
\end{align*}
So $u$ solves a \emph{diagonal} system. Its behavior depends on the
real and complex parts of the eigenvalues.
\begin{enumerate}
\item Positive real $\Rightarrow$ exponential growth.
\item Negative real $\Rightarrow$ exponential decay.
\item Positive imaginary$\Rightarrow$ counter-clockwise rotation.
\item Negative imaginary$\Rightarrow$ clockwise rotation.
\end{enumerate}
